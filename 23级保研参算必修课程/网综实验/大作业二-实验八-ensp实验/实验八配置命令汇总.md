## 配置命令

### RTA

```
<Huawei>system-view
[Huawei]sysname RTA
[RTA]ipv6

[RTA]interface GigabitEthernet 0/0/0
[RTA-GigabitEthernet0/0/0]ipv6 enable
[RTA-GigabitEthernet0/0/0]ipv6 address 2000::1 64
[RTA-GigabitEthernet0/0/0]undo shutdown
[RTA-GigabitEthernet0/0/0]quit

[RTA]interface GigabitEthernet 0/0/1
[RTA-GigabitEthernet0/0/1]ipv6 enable
[RTA-GigabitEthernet0/0/1]ipv6 address 2001::1 64
[RTA-GigabitEthernet0/0/1]undo shutdown
[RTA-GigabitEthernet0/0/1]quit

[RTA]ripng 1
[RTA-ripng-1]quit
[RTA]interface GigabitEthernet 0/0/0
[RTA-GigabitEthernet0/0/0]ripng 1 enable
[RTA-GigabitEthernet0/0/0]quit
[RTA]interface GigabitEthernet 0/0/1
[RTA-GigabitEthernet0/0/1]ripng 1 enable
[RTA-GigabitEthernet0/0/1]quit

[RTA]ripng 1
[RTA-ripng-1]import-route direct
[RTA-ripng-1]quit
```

### RTB

```
<Huawei>system-view
[Huawei]sysname RTB
[RTB]ipv6

[RTB]interface GigabitEthernet 0/0/0
[RTB-GigabitEthernet0/0/0]ipv6 enable
[RTB-GigabitEthernet0/0/0]ipv6 address 2001::2 64
[RTB-GigabitEthernet0/0/0]undo shutdown
[RTB-GigabitEthernet0/0/0]quit

[RTB]interface GigabitEthernet 0/0/1
[RTB-GigabitEthernet0/0/1]ipv6 enable
[RTB-GigabitEthernet0/0/1]ipv6 address 2003::1 64
[RTB-GigabitEthernet0/0/1]undo shutdown
[RTB-GigabitEthernet0/0/1]quit

[RTB]interface GigabitEthernet 0/0/2
[RTB-GigabitEthernet0/0/2]ipv6 enable
[RTB-GigabitEthernet0/0/2]ipv6 address 2002::1 64
[RTB-GigabitEthernet0/0/2]undo shutdown
[RTB-GigabitEthernet0/0/2]quit

# RIPng
[RTB]ripng 1
[RTB-ripng-1]quit
[RTB]interface GigabitEthernet 0/0/0
[RTB-GigabitEthernet0/0/0]ripng 1 enable
[RTB-GigabitEthernet0/0/0]quit

# OSPFv3
[RTB]ospfv3 1
[RTB-ospfv3-1]router-id 2.2.2.2
[RTB-ospfv3-1]quit
[RTB]interface GigabitEthernet 0/0/2
[RTB-GigabitEthernet0/0/2]ospfv3 1 area 0
[RTB-GigabitEthernet0/0/2]quit

# 静态路由
[RTB]ipv6 route-static 2004:: 64 2003::2
[RTB]ipv6 route-static 2005:: 64 2003::2

# 路由引入
[RTB]ripng 1
[RTB-ripng-1]import-route ospfv3 1
[RTB-ripng-1]import-route static
[RTB-ripng-1]quit

[RTB]ospfv3 1
[RTB-ospfv3-1]import-route ripng 1
[RTB-ospfv3-1]import-route static
[RTB-ospfv3-1]quit

[RTB]ripng 1
[RTB-ripng-1]import-route direct
[RTB-ripng-1]quit
```
